\chapter{Spis skrótów i symboli}

\section*{Skróty}

\begin{itemize}
\item[API] interfejs programistyczny aplikacji (\english{Application Programming Interface})
\item[BERT] dwukierunkowe reprezentacje z~transformerów (\english{Bidirectional Encoder Representations from Transformers})
\item[BLIK] natychmiastowy klucz na~poziomie bankowym (\english{Bank Level Immediate Key})
\item[C2C] konsument do konsumenta (\english{consumer-to-consumer})
\item[CPU] jednostka centralna procesora (\english{Central Processing Unit})
\item[FN] fałszywie negatywny (\english{false negative})
\item[FP] fałszywie pozytywny (\english{false positive})
\item[GPU] procesor graficzny (\english{Graphics Processing Unit})
\item[HerBERT] polskojęzyczny model językowy (ang. \english{Herbert BERT}, Allegro)
\item[HTTP] protokół przesyłania hipertekstu (\english{Hypertext Transfer Protocol})
\item[IBAN] międzynarodowy numer rachunku bankowego (\english{International Bank Account Number})
\item[ML] uczenie maszynowe (\english{machine learning})
\item[NLP] przetwarzanie języka naturalnego (\english{Natural Language Processing})
\item[PII] dane osobowe umożliwiające identyfikację (\english{Personally Identifiable Information})
\item[REST] reprezentacyjny transfer stanu (\english{Representational State Transfer})
\item[SQL] strukturalny język zapytań (\english{Structured Query Language})
\item[TP] prawdziwie pozytywny (\english{true positive})
\item[XLM-RoBERTa] wielojęzyczny model RoBERTa (\english{Cross-lingual Language Model RoBERTa})
\end{itemize}

\section*{Symbole matematyczne}

\begin{itemize}
\item[$F_{1,c}$] miara~$F_1$ dla klasy~$c$
\item[$FN_c$] liczba fałszywie negatywnych dla klasy~$c$
\item[$FP_c$] liczba fałszywie pozytywnych dla klasy~$c$
\item[$|C|$] liczba klas w~problemie klasyfikacji
\item[$\overline{F_1}$] makro-uśredniona miara~$F_1$
\item[$P_c$] precyzja dla klasy~$c$
\item[$R_c$] pełność dla klasy~$c$
\item[$TP_c$] liczba prawdziwie pozytywnych dla klasy~$c$
\end{itemize}
